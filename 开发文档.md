  name: string
  uuid: string
  hide: boolean
  hideWhenMouse: boolean
  hideWhenGamepad: boolean

// 按钮类型
interface ZL2NormalButton {
  text: ZL2TranslatableString
  uuid: string
  position: { x: number; y: number }
  buttonSize: ZL2ButtonSize
  buttonStyle: string | null
  visibilityType: 'always' | 'in_game' | 'in_menu'
  clickEvents: ZL2ClickEvent[]
  isSwipple: boolean
  isPenetrable: boolean
  isToggleable: boolean
}

## 🔄 转换流程

### 整体流程

```
用户输入 FCL JSON
    ↓
JSON.parse() 解析
    ↓
类型验证
    ↓
FCLToZL2Converter.convert()
    ├─ convertInfo()        → 转换基本信息
    ├─ convertLayers()      → 转换控件层
    │   ├─ convertButton()  → 转换普通按钮
    │   └─ convertDirectionsToButtons() → 转换方向键
    └─ convertStyles()      → 转换样式
    ↓
生成 ZL2 JSON
    ↓
JSON.stringify() 格式化
    ↓
输出结果
```

### 详细转换步骤

#### 1. 信息转换

```typescript
convertInfo(fcl: FCLController): ZL2Info {
  return {
    name: { default: fcl.name, matchQueue: [] },
    author: { default: fcl.author, matchQueue: [] },
    description: { default: fcl.description, matchQueue: [] },
    versionCode: fcl.versionCode,
    versionName: fcl.version
  }
}
```

#### 2. 控件层转换

```typescript
convertLayers(fcl: FCLController): ZL2Layer[] {
  return fcl.viewGroups.map(group => ({
    name: group.name,
    uuid: generateUUID(),
    hide: group.visibility === 'INVISIBLE',
    hideWhenMouse: false,
    hideWhenGamepad: false,
    visibilityType: 'always',
    normalButtons: [
      ...group.viewData.buttonList.map(btn => convertButton(btn)),
      ...convertDirectionsToButtons(group.viewData.directionList)
    ],
    textBoxes: []
  }))
}
```

#### 3. 按钮转换

```typescript
convertButton(fclBtn: FCLButton): ZL2NormalButton {
  return {
    text: { default: fclBtn.text, matchQueue: [] },
    uuid: generateUUID(),
    position: {
      x: fclBtn.baseInfo.xPosition * 10,  // 坐标转换
      y: fclBtn.baseInfo.yPosition * 10
    },
    buttonSize: convertButtonSize(fclBtn.baseInfo),
    buttonStyle: getStyleUUID(fclBtn.style),
    visibilityType: convertVisibilityType(fclBtn.baseInfo.visibilityType),
    clickEvents: convertButtonEvents(fclBtn.event),
    isSwipple: false,
    isPenetrable: fclBtn.event.pointerFollow,
    isToggleable: fclBtn.event.pressEvent.autoKeep
  }
}
```

#### 4. 方向键转换

```typescript
convertDirectionsToButtons(directions: FCLDirection[]): ZL2NormalButton[] {
  const buttons: ZL2NormalButton[] = []
  
  directions.forEach(dir => {
    // 创建 8 个方向按钮
    const dirButtons = [
      { text: '◤', keys: [upKeycode, leftKeycode] },
      { text: '▲', keys: [upKeycode] },
      { text: '◥', keys: [upKeycode, rightKeycode] },
      // ... 其他方向
    ]
    
    dirButtons.forEach(btn => {
      buttons.push(createDirectionButton(btn))
    })
  })
  
  return buttons
}
```

#### 5. 事件转换

```typescript
convertButtonEvents(event: FCLButtonEvent): ZL2ClickEvent[] {
  const events: ZL2ClickEvent[] = []
  
  // 键码事件
  event.pressEvent.outputKeycodes.forEach(keycode => {
    const glfwKey = FCL_TO_GLFW_KEYMAP[keycode]
    events.push({
      type: glfwKey.startsWith('launcher.event.') ? 'launcher_event' : 'key',
      key: glfwKey
    })
  })
  
  // 文本事件
  if (event.pressEvent.outputText) {
    events.push({
      type: 'send_text',
      key: event.pressEvent.outputText
    })
  }
  
  // 视图组切换
  event.pressEvent.bindViewGroup.forEach(groupId => {
    events.push({
      type: 'switch_layer',
      key: groupId
    })
  })
  
  return events
}
```

#### 6. 样式转换

```typescript
convertStyles(fcl: FCLController): ZL2ButtonStyle[] {
  return fcl.buttonStyles.map(style => ({
    name: style.name,
    uuid: generateUUID(),
    animateSwap: false,
    lightStyle: {
      alpha: calculateAlpha(style.fillColor),
      backgroundColor: SAFE_ZL2_COLORS.TRANSPARENT_BLACK,
      contentColor: SAFE_ZL2_COLORS.WHITE,
      borderWidth: style.strokeWidth / 10,
      borderRadius: {
        topStart: style.cornerRadius / 10,
        topEnd: style.cornerRadius / 10,
        bottomEnd: style.cornerRadius / 10,
        bottomStart: style.cornerRadius / 10
      },
      // ... 其他属性
    },
    darkStyle: { /* 同 lightStyle */ }
  }))
}
```
